<script>

  // ── CONFIGURAÇÃO ────────────────────────────────────────────
  // Troque pela URL da sua Cloud Function depois do deploy
  const API_BASE = "https://goldtech-fabricacoes-api.onrender.com";

  let _subtipoDados = [];
  let _chartVendas  = null;
  let _chartPedras  = null;
  let _autoRefresh  = null;

  // ── UTILITÁRIOS ─────────────────────────────────────────────
  const fmt = {
    num:     (n) => Number(n || 0).toLocaleString("pt-BR"),
    moeda:   (n) => Number(n || 0).toLocaleString("pt-BR", { style: "currency", currency: "BRL", maximumFractionDigits: 0 }),
    moedaK:  (n) => {
      const v = Number(n || 0);
      return v >= 1000 ? "R$" + (v / 1000).toFixed(0) + "k" : fmt.moeda(v);
    },
    data:    (d) => d ? new Date(d).toLocaleDateString("pt-BR") : "—",
    mes:     (s) => {
      if (!s) return "—";
      const [y, m] = s.split("-");
      return new Date(y, m - 1).toLocaleDateString("pt-BR", { month: "short", year: "2-digit" });
    },
  };

  function diasClass(d) {
    if (d <= 15) return "low";
    if (d <= 30) return "mid";
    if (d <= 60) return "high";
    return "danger";
  }

  // ── FETCH ────────────────────────────────────────────────────
  async function apiFetch(endpoint) {
    const r = await fetch(`${API_BASE}/${endpoint}`);
    if (!r.ok) throw new Error(`HTTP ${r.status}`);
    return r.json();
  }

  // ── RENDER RESUMO ────────────────────────────────────────────
  function renderResumo(d) {
    document.getElementById("fab-kpi-estoque").textContent    = fmt.num(d.estoque);
    document.getElementById("fab-kpi-fabricacao").textContent = fmt.num(d.em_fabricacao);
    document.getElementById("fab-kpi-vendidos").textContent   = fmt.num(d.vendidos);
    document.getElementById("fab-kpi-vendidos-mes").textContent = `${fmt.num(d.vendidos_mes)} vendidos este mês`;
    document.getElementById("fab-kpi-ticket").textContent     = fmt.moedaK(d.ticket_medio);
    document.getElementById("fab-kpi-faturamento").textContent = fmt.moedaK(d.faturamento_mes);
  }

  // ── RENDER ALERTAS ───────────────────────────────────────────
  function renderAlertas(lista) {
    const el = document.getElementById("fab-alertas-lista");
    document.getElementById("fab-badge-alertas").textContent = lista.length;

    if (!lista.length) {
      el.innerHTML = '<div class="fab-empty">✅ Nenhum alerta no momento</div>';
      return;
    }

    el.innerHTML = lista.map(a => `
      <div class="fab-alerta ${a.status_alerta}">
        <div>
          <div class="fab-alerta-nome">${a.subtipo}</div>
          <div class="fab-alerta-info">
            ${fmt.num(a.estoque)} em estoque · ${fmt.num(a.vendidos_90d)} vendidos/90d
            ${a.em_fabricacao > 0 ? ` · ${fmt.num(a.em_fabricacao)} em fab.` : ""}
          </div>
        </div>
        <span class="fab-badge ${a.status_alerta}">${a.status_alerta}</span>
      </div>
    `).join("");
  }

  // ── RENDER EM FABRICAÇÃO ─────────────────────────────────────
  function renderFabricacao(lista) {
    const el = document.getElementById("fab-fab-lista");
    document.getElementById("fab-badge-fab").textContent = lista.length;

    if (!lista.length) {
      el.innerHTML = '<div class="fab-empty">Nenhuma peça em fabricação</div>';
      return;
    }

    el.innerHTML = lista.map(p => `
      <div class="fab-fab-item">
        <div>
          <div class="fab-fab-ref">${p.referencia}</div>
          <div class="fab-fab-sub">${p.subtipo || "—"}</div>
          ${p.tipo_pedra ? `<div class="fab-fab-pedra">${p.tipo_pedra}</div>` : ""}
        </div>
        <span class="fab-dias ${diasClass(p.dias)}">${p.dias}d</span>
      </div>
    `).join("");
  }

  // ── RENDER SUBTIPO TABLE ──────────────────────────────────────
  function renderSubtipoTabela(filtro = "todos") {
    const tbody = document.getElementById("fab-subtipo-tbody");
    let dados = [..._subtipoDados].sort((a, b) => Number(b.vendidos) - Number(a.vendidos));

    if (filtro === "criticos") dados = dados.filter(d => Number(d.estoque) <= 3 && Number(d.vendidos) > 0);
    if (filtro === "fabricando") dados = dados.filter(d => Number(d.em_fabricacao) > 0);

    if (!dados.length) {
      tbody.innerHTML = `<tr><td colspan="6"><div class="fab-empty">Nenhum registro</div></td></tr>`;
      return;
    }

    const maxVend = Math.max(...dados.map(d => Number(d.vendidos)), 1);

    tbody.innerHTML = dados.map(d => {
      const est  = Number(d.estoque);
      const fab  = Number(d.em_fabricacao);
      const vend = Number(d.vendidos);
      const pct  = Math.round((est / (est + fab + 0.01)) * 100);
      const lowStock = est <= 3 && vend > 0;

      return `
        <tr>
          <td style="font-weight:600${lowStock ? ";color:var(--fab-orange)" : ""}">
            ${lowStock ? "⚠ " : ""}${d.subtipo}
          </td>
          <td class="right" style="font-weight:700;color:${est === 0 ? "var(--fab-red)" : est <= 3 ? "var(--fab-orange)" : "var(--fab-accent)"}">
            ${fmt.num(est)}
          </td>
          <td>
            <div class="fab-bar-wrap">
              <div class="fab-bar-bg">
                <div class="fab-bar-fill estoque" style="width:${pct}%"></div>
              </div>
              <span style="font-size:11px;color:var(--fab-muted);min-width:32px">${pct}%</span>
            </div>
          </td>
          <td class="right" style="color:${fab > 0 ? "var(--fab-yellow)" : "var(--fab-muted)"}">
            ${fmt.num(fab)}
          </td>
          <td class="right">
            <div class="fab-bar-wrap" style="justify-content:flex-end">
              <div class="fab-bar-bg" style="max-width:80px">
                <div class="fab-bar-fill vendidos" style="width:${Math.round((vend/maxVend)*100)}%"></div>
              </div>
              <span class="fab-num">${fmt.num(vend)}</span>
            </div>
          </td>
          <td class="right" style="color:var(--fab-muted)">
            ${d.ticket_medio ? fmt.moeda(d.ticket_medio) : "—"}
          </td>
        </tr>
      `;
    }).join("");
  }

  window.fabFiltrarSubtipo = function (filtro, tabEl) {
    document.querySelectorAll(".fab-tab").forEach(t => t.classList.remove("active"));
    tabEl.classList.add("active");
    renderSubtipoTabela(filtro);
  };

  // ── RENDER GRÁFICO VENDAS MÊS ────────────────────────────────
  function renderChartVendas(dados) {
    const ctx = document.getElementById("fab-chart-vendas").getContext("2d");
    if (_chartVendas) _chartVendas.destroy();

    _chartVendas = new Chart(ctx, {
      type: "bar",
      data: {
        labels: dados.map(d => fmt.mes(d.mes)),
        datasets: [
          {
            label: "Qtd vendida",
            data: dados.map(d => Number(d.qtd)),
            backgroundColor: "rgba(79,110,247,.7)",
            borderRadius: 6,
            yAxisID: "y",
          },
          {
            label: "Faturamento",
            data: dados.map(d => Number(d.total)),
            type: "line",
            borderColor: "#22c55e",
            backgroundColor: "rgba(34,197,94,.08)",
            tension: 0.4,
            fill: true,
            pointRadius: 3,
            yAxisID: "y1",
          },
        ],
      },
      options: {
        responsive: true,
        interaction: { mode: "index" },
        plugins: {
          legend: { labels: { color: "#9ca3af", font: { size: 11 } } },
          tooltip: {
            callbacks: {
              label: (ctx) =>
                ctx.datasetIndex === 1
                  ? " " + fmt.moeda(ctx.raw)
                  : ` ${ctx.raw} peças`,
            },
          },
        },
        scales: {
          x: { ticks: { color: "#6b7280", font: { size: 11 } }, grid: { color: "rgba(255,255,255,.04)" } },
          y: {
            ticks: { color: "#6b7280", font: { size: 11 } },
            grid: { color: "rgba(255,255,255,.04)" },
            position: "left",
          },
          y1: {
            ticks: {
              color: "#22c55e",
              font: { size: 11 },
              callback: (v) => fmt.moedaK(v),
            },
            grid: { display: false },
            position: "right",
          },
        },
      },
    });
  }

  // ── RENDER GRÁFICO PEDRAS ────────────────────────────────────
  function renderChartPedras(dados) {
    const ctx = document.getElementById("fab-chart-pedras").getContext("2d");
    if (_chartPedras) _chartPedras.destroy();

    const cores = [
      "#4f6ef7","#22c55e","#f59e0b","#ef4444","#a855f7",
      "#06b6d4","#f97316","#ec4899","#84cc16","#14b8a6",
    ];

    _chartPedras = new Chart(ctx, {
      type: "bar",
      data: {
        labels: dados.map(d => d.tipo_pedra),
        datasets: [{
          label: "Vendidos",
          data: dados.map(d => Number(d.qtd)),
          backgroundColor: dados.map((_, i) => cores[i % cores.length] + "cc"),
          borderRadius: 6,
        }],
      },
      options: {
        indexAxis: "y",
        responsive: true,
        plugins: {
          legend: { display: false },
          tooltip: {
            callbacks: {
              label: (ctx) => ` ${ctx.raw} vendidos · TM ${fmt.moeda(dados[ctx.dataIndex].ticket_medio)}`,
            },
          },
        },
        scales: {
          x: { ticks: { color: "#6b7280", font: { size: 11 } }, grid: { color: "rgba(255,255,255,.04)" } },
          y: { ticks: { color: "#9ca3af", font: { size: 11 } }, grid: { display: false } },
        },
      },
    });
  }

  // ── LOAD TODOS OS DADOS ───────────────────────────────────────
  async function fabLoad() {
    const btn = document.getElementById("fab-btn-refresh");
    btn.classList.add("spinning");

    try {
      const [resumo, alertas, fabricacao, subtipo, vendasMes, vendasPedra] = await Promise.all([
        apiFetch("resumo"),
        apiFetch("alertas"),
        apiFetch("em-fabricacao"),
        apiFetch("estoque-subtipo"),
        apiFetch("vendas-mes"),
        apiFetch("vendas-pedra"),
      ]);

      renderResumo(resumo);
      renderAlertas(alertas);
      renderFabricacao(fabricacao);

      _subtipoDados = subtipo;
      renderSubtipoTabela("todos");

      renderChartVendas(vendasMes);
      renderChartPedras(vendasPedra);

      document.getElementById("fab-last-update").textContent =
        "Atualizado: " + new Date().toLocaleTimeString("pt-BR");

    } catch (err) {
      console.error("Fabricações — erro ao carregar:", err);
      document.getElementById("fab-last-update").textContent = "⚠ Erro ao carregar dados";
    } finally {
      btn.classList.remove("spinning");
    }
  }

  // ── REFRESH MANUAL ───────────────────────────────────────────
  window.fabRefresh = fabLoad;

  // ── AUTO-REFRESH A CADA 60s ──────────────────────────────────
  function fabStartAutoRefresh() {
    if (_autoRefresh) clearInterval(_autoRefresh);
    _autoRefresh = setInterval(fabLoad, 60_000);
  }

  // ── INICIALIZAR QUANDO NAVEGAR PARA FABRICAÇÕES ──────────────
  // Se o SalesFlow usa goTo(), conecte aqui:
  const _origGoTo = window.goTo;
  if (typeof _origGoTo === "function") {
    window.goTo = function (page) {
      _origGoTo(page);
      if (page === "fabricacoes") {
        fabLoad();
        fabStartAutoRefresh();
      } else {
        if (_autoRefresh) clearInterval(_autoRefresh);
      }
    };
  } else {
    // Fallback: inicializa direto se a página já estiver visível
    if (document.getElementById("page-fabricacoes")?.style.display !== "none") {
      fabLoad();
      fabStartAutoRefresh();
    }
  }

</script>
